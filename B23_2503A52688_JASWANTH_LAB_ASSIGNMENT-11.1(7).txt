| Event Registration | Queue | First-come, first-served. |

| Bus Scheduling | Priority Queue | Serves urgent buses first. |
| Cafeteria Orders | Queue | First-in, first-out service. |

from collections import deque


class CafeteriaOrderQueue:
    """A simple first-in, first-out queue for cafeteria orders."""

    def __init__(self):
        self.orders = deque()

    def add_order(self, order):
        """Add an order to the back of the queue."""
        self.orders.append(order)

    def serve_order(self):
        """Remove and return the oldest order, or None if empty."""
        if self.orders:
            return self.orders.popleft()
        return None

    def is_empty(self):
        return len(self.orders) == 0


# Small example
if __name__ == "__main__":
    cafeteria = CafeteriaOrderQueue()
    cafeteria.add_order("Order 1: Sandwich")
    cafeteria.add_order("Order 2: Salad")
    cafeteria.add_order("Order 3: Juice")

    print("Serving cafeteria orders:")
    while not cafeteria.is_empty():
        print(cafeteria.serve_order())
