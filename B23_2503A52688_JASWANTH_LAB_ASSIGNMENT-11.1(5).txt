import heapq


class PriorityQueue:
    """A queue where the highest priority item comes out first."""

    def __init__(self):
        """Create an empty queue."""
        self.queue = []
        self.count = 0

    def enqueue(self, item, priority):
        """Add an item with a priority."""
        heapq.heappush(self.queue, (-priority, self.count, item))
        self.count += 1

    def dequeue(self):
        """Remove and return the highest priority item."""
        if not self.queue:
            return None
        return heapq.heappop(self.queue)[2]

    def display(self):
        """Display the items in priority order."""
        for priority, _, item in sorted(self.queue):
            print(item, "(priority:", -priority, ")")


if __name__ == "__main__":
    priority_queue = PriorityQueue()
    priority_queue.enqueue("Low priority", 1)
    priority_queue.enqueue("High priority", 3)
    priority_queue.enqueue("Medium priority", 2)

    priority_queue.display()
    print("Dequeued:", priority_queue.dequeue())
