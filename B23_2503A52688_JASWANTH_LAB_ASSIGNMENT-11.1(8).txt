from collections import deque


class ShoppingCart:
    """Manage a shopping cart using a stack for recent-item undo behavior."""

    def __init__(self):
        self.items = []

    def add_item(self, item):
        """Add a product to the cart."""
        self.items.append(item)

    def remove_last_item(self):
        """Remove and return the most recently added item, or None if empty."""
        if not self.items:
            return None
        return self.items.pop()

    def view_cart(self):
        """Return a copy of current cart items."""
        return list(self.items)


class OrderQueue:
    """Process customer orders in first-in, first-out order."""

    def __init__(self):
        self.orders = deque()

    def add_order(self, order):
        """Add an order to the back of the queue."""
        self.orders.append(order)

    def process_order(self):
        """Remove and return the oldest order, or None if empty."""
        return self.orders.popleft() if self.orders else None


class ProductPriorityQueue:
    """Return the highest-selling product using a max-priority queue."""

    def __init__(self):
        self.products = []

    def enqueue(self, product, sales):
        """Insert a product with its sales count."""
        self.products.append((sales, product))
        self.products.sort(reverse=True)

    def get_top_product(self):
        """Return the product with the highest sales count, or None if empty."""
        if not self.products:
            return None
        _, product = self.products[0]
        return product

    def pop_top_product(self):
        """Remove and return the highest-selling product."""
        if not self.products:
            return None
        _, product = self.products.pop(0)
        return product


class ProductNode:
    """A node in a BST for product search."""

    def __init__(self, product_id, product_name):
        self.product_id = product_id
        self.product_name = product_name
        self.left = None
        self.right = None


class ProductBST:
    """Store products in sorted order by product ID."""

    def __init__(self):
        self.root = None

    def insert(self, product_id, product_name):
        """Insert a product into the BST."""
        if self.root is None:
            self.root = ProductNode(product_id, product_name)
            return

        current = self.root
        while True:
            if product_id < current.product_id:
                if current.left is None:
                    current.left = ProductNode(product_id, product_name)
                    return
                current = current.left
            elif product_id > current.product_id:
                if current.right is None:
                    current.right = ProductNode(product_id, product_name)
                    return
                current = current.right
            else:
                current.product_name = product_name
                return

    def search(self, product_id):
        """Search for a product by product ID."""
        current = self.root
        while current is not None:
            if product_id == current.product_id:
                return current.product_name
            if product_id < current.product_id:
                current = current.left
            else:
                current = current.right
        return None

    def inorder(self):
        """Return products in sorted order."""
        result = []

        def walk(node):
            if node is not None:
                walk(node.left)
                result.append((node.product_id, node.product_name))
                walk(node.right)

        walk(self.root)
        return result


class DeliveryStop:
    """A stop in the delivery route."""

    def __init__(self, location):
        self.location = location
        self.next = None


class DeliveryRoute:
    """Maintain delivery stops in order using a linked list."""

    def __init__(self):
        self.head = None

    def add_stop(self, location):
        """Add a delivery stop to the end of the route."""
        new_stop = DeliveryStop(location)
        if self.head is None:
            self.head = new_stop
            return

        current = self.head
        while current.next is not None:
            current = current.next
        current.next = new_stop

    def show_route(self):
        """Return the delivery stops in order."""
        stops = []
        current = self.head
        while current is not None:
            stops.append(current.location)
            current = current.next
        return stops
